cmake_minimum_required(VERSION 3.22)
project(mansion_binding_probe LANGUAGES C CXX)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "Pinned local engine API")
set(GODOTCPP_PRECISION "single" CACHE STRING "Engine precision")
set(GODOTCPP_TARGET "template_debug" CACHE STRING "Binding target")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "Minimal API")
add_subdirectory("../../tools/godot-cpp" godot-cpp)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server)
add_library(mansion_runtime STATIC ../../src/compositor-core.cpp ../../src/compositor-runtime.cpp)
set_target_properties(mansion_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(mansion_runtime PUBLIC PkgConfig::WAYLAND_SERVER)
add_library(mansion_probe SHARED ../../src/godot/binding_probe.cpp ../../src/godot/compositor_session.cpp)
target_link_libraries(mansion_probe PRIVATE godot::cpp mansion_runtime)
target_compile_options(mansion_probe PRIVATE -Wall -Wextra)
target_link_options(mansion_probe PRIVATE -Wl,--no-undefined)
set_target_properties(mansion_probe PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
pkg_check_modules(WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
add_executable(runtime-client ../runtime-client.c)
target_link_libraries(runtime-client PRIVATE PkgConfig::WAYLAND_CLIENT)
add_executable(runtime-lifecycle ../test_compositor_runtime.cpp)
target_include_directories(runtime-lifecycle PRIVATE ../../src)
target_link_libraries(runtime-lifecycle PRIVATE mansion_runtime)
